class Solution:
    def lengthOfLongestSubstring(self, s: str) -> int:
        map={}
        i=j=0
        maxlen=0
        while j<len(s):
            if map.get(s[j]) is not None:
                if map[s[j]] >= i:
                    i = map[s[j]] + 1
    

                

            map[s[j]]=j
            maxlen=max(maxlen,j-i+1)
            j=j+1
        
        return maxlen
