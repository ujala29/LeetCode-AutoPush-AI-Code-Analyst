class Solution:
    def longestOnes(self, nums: List[int], k: int) -> int:

        zeros=0
        i=j=0
        maxcount=0
        while j<len(nums):
            if nums[j]==0:
                zeros+=1
            
            while zeros>k:
                if nums[i]==0:
                    zeros-=1
                
                i+=1
            
            maxcount=max(maxcount,j-i+1)
            j+=1
        return maxcount
