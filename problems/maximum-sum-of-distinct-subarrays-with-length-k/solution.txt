class Solution:
    def maximumSubarraySum(self, nums: List[int], k: int) -> int:
        
        sum = 0
        maxsum = 0
        s = set()

        i = 0
        j = 0

        while j < len(nums):

            # Step 1: remove duplicates
            while nums[j] in s:
                s.remove(nums[i])
                sum -= nums[i]
                i += 1

            # Step 2: add current element
            s.add(nums[j])
            sum += nums[j]

            # Step 3: check window size
            if j - i + 1 == k:
                maxsum = max(maxsum, sum)

                # slide window
                s.remove(nums[i])
                sum -= nums[i]
                i += 1

            j += 1

        return maxsum
